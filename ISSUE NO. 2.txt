#region PORTION FOR :: VALIDATE & RESOLVE ProductCombinationId (CRITICAL)
// Validate line items and resolve ProductCombinationId from Attribute
foreach (var item in postedData.PostedDataAFInvoicePPI)
{
    if (item.ProductId <= 0)
    {
        await transaction.RollbackAsync();
        return ServiceResult.failure(
            "Invalid product in line item. ProductId must be > 0.",
            (int)Code.BadRequest
        );
    }

    // If ProductCombinationId not provided, try to resolve from Attribute
    if ((item.ProductCombinationId == null || item.ProductCombinationId <= 0) 
        && !string.IsNullOrEmpty(item.Attribute))
    {
        try
        {
            item.ProductCombinationId = await _commonServices.get_productCombination(
                item.ProductId,
                item.Attribute
            );
        }
        catch (Exception ex)
        {
            await transaction.RollbackAsync();
            return ServiceResult.failure(
                $"Failed to resolve product variant for: {item.Attribute}. Error: {ex.Message}",
                (int)Code.BadRequest
            );
        }
    }

    // Validate ProductCombinationId is now valid
    if (item.ProductCombinationId == null || item.ProductCombinationId <= 0)
    {
        // Check if product has variants
        var hasVariants = await _eRPOSContext.osvProductCombination
            .AnyAsync(x => x.ProductId == item.ProductId && x.Status == true);

        if (hasVariants)
        {
            await transaction.RollbackAsync();
            return ServiceResult.failure(
                $"Product has variants but no variant selected. Please select a variant.",
                (int)Code.BadRequest
            );
        }
        // If no variants, OK to proceed with ProductCombinationId = 0
    }
}
#endregion
