
    SELECT
        IL.[TransactionDate],
        IL.[Code]                                                          AS [LedgerBatchCode],

        -- Movement type — quantity direction is the source of truth, never inferred from RefDocumentType alone
        CASE WHEN IL.[QuantityIn]  > 0 THEN 'Stock In'
             WHEN IL.[QuantityOut] > 0 THEN 'Stock Out'
             ELSE 'N/A' END                                                AS [MovementType],

        -- Originating document — resolved by type, never guessed
        CASE IL.[RefDocumentType]
            WHEN 22 THEN 'Bill'
            WHEN 16 THEN 'Invoice'
            WHEN 27 THEN 'Adjustment'
            ELSE 'Other' END                                              AS [DocumentType],
        COALESCE(B.[Code], I.[Code], A.[Code])                             AS [RefDocumentCode],
        IL.[RefDocumentId],

        -- Product / variant
        P.[Description]                                                   AS [Product],
        CASE WHEN IL.[ProductCombinationId] is null THEN 'N/A'
             ELSE CCE.[Description] END                                    AS [Variant],

        B_ACB.[Description]                                                AS [Location],
        IL.[Batch],
        IL.[ExpiryDate],

        -- Movement quantities + running balance (per product/variant/location, chronological)
        IL.[QuantityIn],
        IL.[QuantityOut],
        SUM(IL.[QuantityIn] - IL.[QuantityOut]) OVER (
            PARTITION BY IL.[ProductId], IL.[ProductCombinationId], IL.[LocationId]
            ORDER BY IL.[TransactionDate], IL.[Id]
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        )                                                                  AS [RunningBalance],

        -- Cost side — from the costing engine, exact, this is what feeds valuation/COGS
        IL.[UnitPurchasePrice]                                             AS [UnitCost],
        IL.[Debit],
        IL.[Credit],

        -- Reference sale rate — informational only, never used for revenue reconciliation (see AFInvoicePPI.ChargedAmount for that)
        IL.[UnitSalePrice],

        -- Open-layer traceability
        IL.[ReceiptBalanceQuantity],           -- meaningful on IN rows: remaining un-consumed balance on that layer
        IL.[ConsumedInventoryLedgerId],         -- meaningful on OUT rows: which IN layer this specific split consumed

        IL.[Description]                                                   AS [Remarks],
        U.[Description]                                                    AS [RecordedBy],
        IL.[GuID]

    FROM        [dbo].[AFInventoryLedger] AS IL WITH (NOLOCK)
    INNER JOIN  [dbo].[IProduct]          AS P    WITH (NOLOCK) ON P.Id = IL.ProductId
    LEFT JOIN   [dbo].[IProductCCE]       AS CCE  WITH (NOLOCK) ON CCE.Id = IL.ProductCombinationId AND IL.ProductCombinationId <> -2
    LEFT JOIN   [dbo].[ACBranch]          AS B_ACB WITH (NOLOCK) ON B_ACB.Id = IL.LocationId
    LEFT JOIN   [dbo].[ACUser]            AS U    WITH (NOLOCK) ON U.Id = IL.CreatedBy
    LEFT JOIN   [dbo].[AFBill]            AS B    WITH (NOLOCK) ON IL.RefDocumentType = 22 AND B.Id = IL.RefDocumentId
    LEFT JOIN   [dbo].[AFInvoice]         AS I    WITH (NOLOCK) ON IL.RefDocumentType = 16 AND I.Id = IL.RefDocumentId
    LEFT JOIN   [dbo].[IAdjustment]       AS A    WITH (NOLOCK) ON IL.RefDocumentType = 27 AND A.Id = IL.RefDocumentId
    ORDER BY
        IL.CreatedOn
